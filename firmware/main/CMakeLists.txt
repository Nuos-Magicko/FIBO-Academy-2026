# Component versions for HELLO, read from the managed components actually used in this build and
# written as constexpr strings into the generated header component_versions.h.
function(read_component_version component_dir out_var)
    file(STRINGS "${CMAKE_SOURCE_DIR}/managed_components/${component_dir}/idf_component.yml" line
         REGEX "^version:")
    string(REGEX REPLACE "^version: *\"?([^\"]*)\"?.*" "\\1" version "${line}")
    set(${out_var} "${version}" PARENT_SCOPE)
endfunction()

read_component_version("espressif__esp-dl" ESP_DL_VERSION)
read_component_version("espressif__esp32-camera" ESP32_CAMERA_VERSION)
read_component_version("espressif__pedestrian_detect" PEDESTRIAN_DETECT_VERSION)

file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/component_versions.h" CONTENT
"/** @file component_versions.h Generated by main/CMakeLists.txt: do not edit. */
#pragma once
constexpr char ESP_DL_VERSION[] = \"@ESP_DL_VERSION@\";
constexpr char ESP32_CAMERA_VERSION[] = \"@ESP32_CAMERA_VERSION@\";
constexpr char PEDESTRIAN_DETECT_VERSION[] = \"@PEDESTRIAN_DETECT_VERSION@\";
" @ONLY)
target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
